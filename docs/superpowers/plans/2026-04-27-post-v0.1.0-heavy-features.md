> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Architecture:** Non-UI changes (model, quiz, vfs, command, data) land first; the TUI rewrite comes last because it restructures how App produces output (println! → Vec<String> returned from handlers). pub fn ls(&self, target: Option<&str>, show_all: bool) -> Result<Vec<String>> {
```rust
    let path = self.resolve_path(target.unwrap_or("."));
    let node = self.node_at(&path)?;
    match node {
        Node::Directory { children } => {
            let mut entries: Vec<String> = children.keys().cloned().collect();
            if show_all {
                entries.insert(0, "..".to_string());
                entries.insert(0, ".".to_string());
            }
            Ok(entries)
        }
        Node::File { .. } => Err(anyhow!("not a directory: {}", display_path(&path))),
    }
}
```

```rust
pub fn rm(&mut self, target: &str) -> Result<()> {
    let path = self.resolve_path(target);
    if matches!(self.node_at(&path)?, Node::Directory { .. }) {
        return Err(anyhow!("is a directory: {} (use rm -r)", target));
    }
    self.remove_node(&path)
}

pub fn rm_recursive(&mut self, target: &str) -> Result<()> {
    let path = self.resolve_path(target);
```

```rust
pub fn mkdir_p(&mut self, target: &str) -> Result<()> {
    let path = self.resolve_path(target);
    for len in 1..=path.len() {
        let partial = path[..len].to_vec();
        if self.node_at(&partial).is_err() {
            self.insert_node(partial, Node::directory())?;
        }
    }
    Ok(())
}
```

```rust
pub fn grep_in_file(&self, needle: &str, file: &str) -> Result<Vec<String>> {
    let path = self.resolve_path(file);
    match self.node_at(&path)? {
        Node::File { content } => Ok(content
            .lines()
            .filter(|line| line.contains(needle))
            .map(|line| format!("{}:{}", file, line))
            .collect()),
        Node::Directory { .. } => Err(anyhow!("is a directory: {}", file)),
    }
}
```

- [ ] **Step 1: Rewrite execute_shell with full flag support**

Replace the entire `execute_shell` function:

```rust
fn execute_shell(tokens: &[String], vfs: &mut VirtualFs) -> Result<ExecOutput> {
    match tokens[0].as_str() {
        "pwd" => Ok(ExecOutput::single(vfs.pwd())),
        "ls" => {
            let mut show_all = false;
            let mut long = false;
            let mut target = None;
            for t in &tokens[1..] {
                if t.starts_with('-') {
                    if t.contains('a') { show_all = true; }
                    if t.contains('l') { long = true; }
                } else {
                    target = Some(t.as_str());
                }
            }
            let lines = vfs.ls(target, show_all)?;
            if long {
                Ok(ExecOutput {
                    stdout: lines.iter().map(|n| format!("  {}", n)).collect(),
                })
            } else {
                Ok(ExecOutput { stdout: lines })
            }
        }
        "cd" => {
            let target = tokens.get(1).ok_or_else(|| anyhow!("cd requires target"))?;
            vfs.cd(target)?;
            Ok(ExecOutput::single(format!("cwd => {}", vfs.pwd())))
        }
        "mkdir" => {
            let parents = tokens.iter().any(|t| t == "-p");
            let target = tokens.iter().skip(1)
                .find(|t| !t.starts_with('-'))
                .ok_or_else(|| anyhow!("mkdir requires target"))?;
            if parents {
                vfs.mkdir_p(target)?;
            } else {
                vfs.mkdir(target)?;
            }
            Ok(ExecOutput::single(format!("created {}", target)))
        }
        "touch" => {
            let target = tokens.get(1).ok_or_else(|| anyhow!("touch requires target"))?;
            vfs.touch(target)?;
            Ok(ExecOutput::single(format!("touched {}", target)))
        }
        "cat" => {
            let target = tokens.get(1).ok_or_else(|| anyhow!("cat requires target"))?;
            Ok(ExecOutput::single(vfs.cat(target)?))
        }
        "cp" => {
            let from = tokens.get(1).ok_or_else(|| anyhow!("cp requires source"))?;
            let to = tokens.get(2).ok_or_else(|| anyhow!("cp requires destination"))?;
            vfs.cp(from, to)?;
            Ok(ExecOutput::single(format!("copied {} -> {}", from, to)))
        }
        "mv" => {
            let from = tokens.get(1).ok_or_else(|| anyhow!("mv requires source"))?;
            let to = tokens.get(2).ok_or_else(|| anyhow!("mv requires destination"))?;
            vfs.mv(from, to)?;
            Ok(ExecOutput::single(format!("moved {} -> {}", from, to)))
        }
        "rm" => {
            let recursive = tokens.iter().any(|t| {
                matches!(t.as_str(), "-r" | "-rf" | "-fr" | "-R")
            });
            let target = tokens.iter().skip(1)
                .find(|t| !t.starts_with('-'))
                .ok_or_else(|| anyhow!("rm requires target"))?;
            if recursive {
                vfs.rm_recursive(target)?;
            } else {
                vfs.rm(target)?;
            }
            Ok(ExecOutput::single(format!("removed {}", target)))
        }
        "find" => {
            let mut name_pattern: Option<String> = None;
            let mut i = 1;
            while i < tokens.len() {
                if tokens[i] == "-name" {
                    if let Some(next) = tokens.get(i + 1) {
                        name_pattern = Some(
                            next.trim_matches('"')
                                .trim_start_matches("./")
                                .to_string(),
                        );
                        i += 2;
                    } else {
                        i += 1;
                    }
                } else {
                    i += 1;
                }
            }
            let name = match &name_pattern {
                Some(n) => n.as_str(),
                None => tokens.last()
                    .map(|s| s.trim_matches('"').trim_start_matches("./"))
                    .unwrap_or(""),
            };
            Ok(ExecOutput { stdout: vfs.find_name(name) })
        }
        "grep" => {
            let args: Vec<&str> = tokens.iter().skip(1)
                .filter(|t| !t.starts_with('-'))
                .map(|s| s.as_str())
                .collect();
            let needle = args.first().ok_or_else(|| anyhow!("grep requires needle"))?;
            if let Some(file) = args.get(1) {
                Ok(ExecOutput { stdout: vfs.grep_in_file(needle, file)? })
            } else {
                Ok(ExecOutput { stdout: vfs.grep(needle) })
            }
        }
        "echo" => Ok(ExecOutput::single(tokens[1..].join(" "))),
        other => Err(anyhow!("unsupported command: {}", other)),
    }
}
```

```rust
"run" => {
    let mut name = None;
    let mut image: Option<&str> = None;
    let mut detach = false;
    let mut cursor = 2usize;
    while cursor < tokens.len() {
        match tokens[cursor].as_str() {
            "--name" => {
                name = tokens.get(cursor + 1).map(String::as_str);
                cursor += 2;
            }
            "-d" | "--detach" => {
                detach = true;
                cursor += 1;
            }
            "-p" | "-e" | "-v" => {
                cursor += 2; // skip flag and its value
            }
            value if !value.starts_with('-') => {
                image = Some(value);
                cursor += 1;
            }
            _ => cursor += 1,
        }
    }
    let image = image.ok_or_else(|| anyhow!("docker run requires image"))?;
    Ok(ExecOutput::single(docker.run(image, name, detach)?))
}
```

```rust
pub fn run(&mut self, image: &str, name: Option<&str>, detach: bool) -> Result<String> {
    let image = normalize_image(image);
    if !self.images.contains(&image) {
        return Err(anyhow!("image not found: {}", image));
    }
    let name = name
        .map(ToString::to_string)
        .unwrap_or_else(|| format!("ctr-{}", self.containers.len() + 1));
    self.containers.push(Container {
        name: name.clone(),
        image: image.clone(),
        running: true,
        logs: vec![format!("container {} started from {}", name, image)],
    });
    let mode = if detach { " (detached)" } else { "" };
    Ok(format!("started {}{}", name, mode))
}
```

```rust
use ratatui::{
    Frame,
    layout::{Constraint, Direction, Layout},
    style::{Color, Modifier, Style},
    widgets::{Block, Borders, List, ListItem, Paragraph, Wrap},
};

pub struct TuiState {
    pub input: String,
    pub cursor_pos: usize,
    pub completions: Vec<String>,
    pub completion_index: Option<usize>,
    pub history: Vec<String>,
    pub prompt_text: String,
    pub status_line: String,
}

impl TuiState {
    pub fn new(prompt_text: String, status_line: String) -> Self {
        Self {
            input: String::new(),
            cursor_pos: 0,
            completions: Vec::new(),
            completion_index: None,
            history: Vec::new(),
            prompt_text,
            status_line,
        }
    }

    pub fn push_history(&mut self, line: impl Into<String>) {
        self.history.push(line.into());
        if self.history.len() > 300 {
            self.history.drain(..100);
        }
    }

    pub fn extend_history(&mut self, lines: Vec<String>) {
        for line in lines {
            self.push_history(line);
        }
    }
}

pub fn render(frame: &mut Frame, state: &TuiState) {
    let completion_height = if state.completions.is_empty() {
        0
    } else {
        (state.completions.len().min(6) as u16) + 2
    };

    let chunks = Layout::default()
        .direction(Direction::Vertical)
        .constraints([
            Constraint::Length(1),
            Constraint::Min(3),
            Constraint::Length(3),
            Constraint::Length(completion_height),
            Constraint::Min(3),
        ])
        .split(frame.area());

    // Status bar
    frame.render_widget(
        Paragraph::new(state.status_line.as_str())
            .style(Style::default().bg(Color::DarkGray).fg(Color::White)),
        chunks[0],
    );

    // Prompt pane
    frame.render_widget(
        Paragraph::new(state.prompt_text.as_str())
            .block(Block::default().title("Question").borders(Borders::ALL))
            .wrap(Wrap { trim: true }),
        chunks[1],
    );

    // Input area
    frame.render_widget(
        Paragraph::new(format!("> {}", state.input).as_str())
            .block(
                Block::default()
                    .title("Input  (Tab: complete  Shift+Tab: prev  Ctrl+C: quit)")
                    .borders(Borders::ALL),
            ),
        chunks[2],
    );

    // Cursor — left border + "> " + offset
    let cursor_x = chunks[2].x + 1 + 2 + state.cursor_pos as u16;
    let cursor_y = chunks[2].y + 1;
    if cursor_x < chunks[2].x + chunks[2].width.saturating_sub(1) {
        frame.set_cursor_position((cursor_x, cursor_y));
    }

    // Completions (only rendered when non-empty)
    if !state.completions.is_empty() && completion_height > 0 {
        let items: Vec<ListItem> = state
            .completions
            .iter()
            .take(6)
            .enumerate()
            .map(|(i, c)| {
                if Some(i) == state.completion_index {
                    ListItem::new(format!("▶ {}", c))
                        .style(Style::default().fg(Color::Yellow).add_modifier(Modifier::BOLD))
                } else {
                    ListItem::new(format!("  {}", c))
                }
            })
            .collect();
        frame.render_widget(
            List::new(items)
                .block(Block::default().title("Completions").borders(Borders::ALL)),
            chunks[3],
        );
    }

    // History pane — show last N lines that fit
    let inner_height = chunks[4].height.saturating_sub(2) as usize;
    let visible: String = state
        .history
        .iter()
        .rev()
        .take(inner_height)
        .collect::<Vec<_>>()
        .into_iter()
        .rev()
        .map(|s| s.as_str())
        .collect::<Vec<_>>()
        .join("\n");
    frame.render_widget(
        Paragraph::new(visible.as_str())
            .block(Block::default().title("History").borders(Borders::ALL))
            .wrap(Wrap { trim: false }),
        chunks[4],
    );
}
```

```rust
fn run_tui(
    &mut self,
    terminal: &mut Terminal<CrosstermBackend<io::Stdout>>,
) -> Result<()> {
    let mut tui = TuiState::new(
        self.state.current_prompt.clone().unwrap_or_default(),
        self.render_status(),
    );

    loop {
        terminal.draw(|f| ui::render(f, &tui))?;

        if event::poll(Duration::from_millis(50))? {
            if let Event::Key(key) = event::read()? {
                if self.handle_key(key, &mut tui)? {
                    break;
                }
            }
        }
    }
    Ok(())
}

fn handle_key(
    &mut self,
    key: ratatui::crossterm::event::KeyEvent,
    tui: &mut TuiState,
) -> Result<bool> {
    match key.code {
        KeyCode::Char('c') if key.modifiers.contains(KeyModifiers::CONTROL) => {
            self.persist_session()?;
            return Ok(true);
        }
        KeyCode::Char('d') if key.modifiers.contains(KeyModifiers::CONTROL) => {
            self.persist_session()?;
            return Ok(true);
        }
        KeyCode::Enter => {
            let line = tui.input.trim().to_string();
            tui.input.clear();
            tui.cursor_pos = 0;
            tui.completions.clear();
            tui.completion_index = None;

            if line.is_empty() {
                return Ok(false);
            }

            tui.push_history(format!("> {}", line));

            let (exit, output) = self.process_line(&line)?;
            tui.extend_history(output);
            tui.status_line = self.render_status();
            tui.prompt_text = self.state.current_prompt.clone().unwrap_or_default();

            if exit {
                return Ok(true);
            }
        }
        KeyCode::Tab => {
            if tui.completions.is_empty() {
                tui.completions = self.generate_completions(&tui.input);
            }
            if !tui.completions.is_empty() {
                let next = tui.completion_index
                    .map(|i| (i + 1) % tui.completions.len())
                    .unwrap_or(0);
                tui.completion_index = Some(next);
                tui.input = tui.completions[next].clone();
                tui.cursor_pos = tui.input.len();
            }
        }
        KeyCode::BackTab => {
            if !tui.completions.is_empty() {
                let prev = tui.completion_index
                    .map(|i| if i == 0 { tui.completions.len() - 1 } else { i - 1 })
                    .unwrap_or(0);
                tui.completion_index = Some(prev);
                tui.input = tui.completions[prev].clone();
                tui.cursor_pos = tui.input.len();
            }
        }
        KeyCode::Char(c) => {
            tui.completions.clear();
            tui.completion_index = None;
            tui.input.insert(tui.cursor_pos, c);
            tui.cursor_pos += c.len_utf8();
        }
        KeyCode::Backspace => {
            if tui.cursor_pos > 0 {
                tui.completions.clear();
                tui.completion_index = None;
                // walk back to a char boundary
                let mut pos = tui.cursor_pos - 1;
                while pos > 0 && !tui.input.is_char_boundary(pos) {
                    pos -= 1;
                }
                tui.cursor_pos = pos;
                tui.input.remove(pos);
            }
        }
        KeyCode::Left => {
            if tui.cursor_pos > 0 {
                let mut pos = tui.cursor_pos - 1;
                while pos > 0 && !tui.input.is_char_boundary(pos) {
                    pos -= 1;
                }
                tui.cursor_pos = pos;
            }
        }
        KeyCode::Right => {
            if tui.cursor_pos < tui.input.len() {
                let mut pos = tui.cursor_pos + 1;
                while pos < tui.input.len() && !tui.input.is_char_boundary(pos) {
                    pos += 1;
                }
                tui.cursor_pos = pos;
            }
        }
        KeyCode::Esc => {
            tui.completions.clear();
            tui.completion_index = None;
        }
        _ => {}
    }
    Ok(false)
}

fn generate_completions(&self, input: &str) -> Vec<String> {
    if self.state.completion == CompletionMode::Off {
        return Vec::new();
    }

    let limit = match self.state.difficulty {
        Difficulty::Easy => 12,
        Difficulty::Normal => 6,
        Difficulty::Hard => 3,
    };

    let start = input.rfind(char::is_whitespace).map(|i| i + 1).unwrap_or(0);
    let current_word = &input[start..];
    let prefix_before = &input[..start];

    let mut candidates: Vec<String> = base_completions()
        .into_iter()
        .filter(|item| item.starts_with(input))
        .collect();

    candidates.extend(
        self.vfs.borrow()
            .find_paths_with_prefix(current_word)
            .into_iter()
            .map(|p| format!("{}{}", prefix_before, p)),
    );
    candidates.extend(
        self.docker.borrow()
            .completions(current_word)
            .into_iter()
            .map(|p| format!("{}{}", prefix_before, p)),
    );

    candidates.sort();
    candidates.dedup();
    candidates.truncate(limit);
    candidates
}
```
